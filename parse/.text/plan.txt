{
	close_all = false
	for (int i = 0; i < max_depth; i++) {
		if close_all
		|| current_rect[depth].dso != sample.dso
		|| current_rect[depth].offset != sample.offset {
			close_all = true
			close(current_depth...)
			open(sample)
		} else {
			update(current_rect[depth], sample[depth]);
		}
	}
}
